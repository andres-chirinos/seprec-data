html`<hr style="margin:.8rem 0;opacity:.3">
<small style="color:#666">
  <b>Tip:</b> Haz clic en un punto para ver el detalle de la empresa.
</small>`
{
  const filtered = geo_data
    .filter(d => filtroEstado === "Todos" || d.estado === filtroEstado)
    .filter(d => filtroTipo  === "Todos" || d.tipo_ue === filtroTipo)
    .slice(0, maxPuntos);

  const colorMap = {
    ACTIVO:    "#198754",
    CANCELADO: "#dc3545",
    PENDIENTE: "#fd7e14",
    INACTIVO:  "#6c757d",
    CREADO:    "#0dcaf0",
    ELIMINADO: "#343a40",
  };

  const div = DOM.element("div");
  div.style.cssText = [
    "height:100%","min-height:480px","width:100%",
    "border-radius:10px","overflow:hidden",
    "box-shadow:0 2px 12px rgba(0,0,0,.15)"
  ].join(";");
  yield div;

  const L = await require("leaflet@1.9.4/dist/leaflet.js");
  if (!document.querySelector("link[href*='leaflet']")) {
    const lnk = document.createElement("link");
    lnk.rel = "stylesheet";
    lnk.href = "https://unpkg.com/leaflet@1.9.4/dist/leaflet.css";
    document.head.appendChild(lnk);
  }
  await new Promise(r => setTimeout(r, 80));

  const map = L.map(div, { center: [-16.5, -64.5], zoom: 6 });
  L.tileLayer("https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png", {
    attribution: "© OpenStreetMap © CARTO", maxZoom: 19
  }).addTo(map);

  filtered.forEach(d => {
    const color = colorMap[d.estado] || "#0d6efd";
    L.circleMarker([d.lat, d.lon], {
      radius: 5, fillColor: color,
      color: "#fff", weight: 0.5,
      opacity: 1, fillOpacity: 0.78,
    })
    .bindPopup(`<div style="font-family:system-ui;min-width:190px;line-height:1.55">
      <b style="font-size:.95rem">${d.nombre || "—"}</b><br>
      <span style="color:#888;font-size:.8rem">${d.tipo_ue || ""}</span>
      <hr style="margin:.35rem 0">
      <b>Estado:</b> <span style="color:${color}">${d.estado}</span><br>
      <b>Matrícula:</b> ${d.estado_matricula}<br>
      <b>Municipio:</b> ${d.municipio || "—"}<br>
      <b>Depto.:</b> ${d.departamento || "—"}
    </div>`)
    .addTo(map);
  });

  const legend = L.control({ position: "bottomright" });
  legend.onAdd = () => {
    const el = L.DomUtil.create("div");
    el.style.cssText = "background:#fff;padding:.7rem .9rem;border-radius:8px;font-size:.78rem;box-shadow:0 2px 8px rgba(0,0,0,.2)";
    el.innerHTML = Object.entries(colorMap)
      .map(([k,v]) => `<div style="display:flex;align-items:center;gap:.35rem;margin:.12rem 0">
        <div style="width:11px;height:11px;border-radius:50%;background:${v}"></div>${k}</div>`)
      .join("") + `<hr style="margin:.35rem 0"><small>${filtered.length.toLocaleString()} empresas</small>`;
    return el;
  };
  legend.addTo(map);
}
